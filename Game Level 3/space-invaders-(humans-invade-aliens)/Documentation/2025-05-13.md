uniform vec4 hit_flash_color : source_color = vec4(1.0, 1.0, 1.0, 1.0);

	// Called for every vertex the material is visible on.

	//sprite para dar un diseño esperado

	vec4 sprite_color = texture(TEXTURE, UV);
